# 灵犀拼音输入法 — Windows 端版本号单一来源
#
# 字段:
#   PRIMARY_VERSION     语义化版本号 (MAJOR.MINOR.PATCH[-PRERELEASE])
#   RELEASE_CHANNEL     通道: stable | beta | dev
#   BUILD_DATE          构建日期 (YYYY-MM-DD, 由 build_win.sh 自动刷新)
#   GIT_COMMIT          提交哈希或 HEAD 占位
#
# 注: Win 端无 versionCode (Windows 安装包按 MS Installer ProductVersion 4 段数字递增,
#     这里 PRIMARY_VERSION 直接拼成 "MAJOR.MINOR.PATCH.BUILD" 给 MSI / NSIS 用)
#
# 例:
#   1.0.0       stable → ProductVersion=1.0.0.0  内部版本号 "1.0.0 stable"
#   1.0.0-beta.1 beta  → ProductVersion=1.0.0.1  内部版本号 "1.0.0-beta.1 beta"
#   1.0.0-dev   dev    → ProductVersion=1.0.0.9  内部版本号 "1.0.0-dev dev"
#
# 升版注意: Win 端 PRIMARY_VERSION 跟随 Android 端同基线 (用户已在 2026-08-29 决策对齐),
#           Win 端历史基线 0.7.0, 升到 1.0.0 表示 "首个跨平台公开版本"。

PRIMARY_VERSION=1.0.0-beta.8
RELEASE_CHANNEL=beta
BUILD_DATE=2026-09-05
GIT_COMMIT=HEAD
